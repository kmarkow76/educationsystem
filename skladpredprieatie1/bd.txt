-- 1. ТАБЛИЦА: Поставщики / Контрагенты
CREATE TABLE suppliers (
    id SERIAL PRIMARY KEY,
    company_name VARCHAR(150) NOT NULL,
    contact_phone VARCHAR(20) NOT NULL,
    is_permanent BOOLEAN DEFAULT FALSE NOT NULL -- Флаг "Постоянный контрагент" для скидки 5%
);

-- 2. ТАБЛИЦА: Сотрудники склада
CREATE TABLE employees (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    position VARCHAR(100) NOT NULL
);

-- 3. ТАБЛИЦА: Товары
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    product_name VARCHAR(150) NOT NULL,
    category VARCHAR(100) NOT NULL,
    unit_of_measure VARCHAR(20) NOT NULL, -- Единица измерения (шт, кг, м)
    quantity_in_stock INT NOT NULL DEFAULT 0 CHECK (quantity_in_stock >= 0), -- Контроль остатка
    unit_price DECIMAL(12, 2) NOT NULL CHECK (unit_price >= 0),
    supplier_id INT NOT NULL,
    CONSTRAINT fk_product_supplier FOREIGN KEY (supplier_id) REFERENCES suppliers(id) ON DELETE CASCADE
);

-- 4. ТАБЛИЦА: Складские операции (Поступления и Выдачи)
CREATE TABLE warehouse_operations (
    id SERIAL PRIMARY KEY,
    product_id INT NOT NULL,
    employee_id INT NOT NULL,
    operation_type VARCHAR(20) NOT NULL CHECK (operation_type IN ('Поступление', 'Выдача')),
    quantity INT NOT NULL CHECK (quantity > 0),
    operation_date DATE NOT NULL DEFAULT CURRENT_DATE,
    recipient_name VARCHAR(150), -- Получатель (заполняется при выдаче со склада)
    CONSTRAINT fk_operation_product FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE,
    CONSTRAINT fk_operation_employee FOREIGN KEY (employee_id) REFERENCES employees(id) ON DELETE CASCADE
);
