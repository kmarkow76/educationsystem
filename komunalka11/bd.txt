-- ============================================================================
-- 1. СОЗДАНИЕ ТАБЛИЦ (Соблюдение 3НФ и ссылочной целостности)
-- ============================================================================

-- Таблица: Коммунальные услуги
CREATE TABLE services (
    id SERIAL PRIMARY KEY,
    service_name VARCHAR(100) NOT NULL,
    tariff NUMERIC(10, 2) NOT NULL CHECK (tariff >= 0)
);

-- Таблица: Жильцы
CREATE TABLE citizens (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    has_privilege BOOLEAN NOT NULL DEFAULT FALSE -- Флаг льготы (для скидки 25%)
);

-- Таблица: Лицевые счета
CREATE TABLE accounts (
    id SERIAL PRIMARY KEY,
    account_number VARCHAR(50) UNIQUE NOT NULL,
    address VARCHAR(250) NOT NULL,
    citizen_id INT NOT NULL,
    CONSTRAINT fk_account_citizen FOREIGN KEY (citizen_id) REFERENCES citizens(id) ON DELETE CASCADE
);

-- Таблица: Показания счетчиков
CREATE TABLE meter_readings (
    id SERIAL PRIMARY KEY,
    account_id INT NOT NULL,
    service_id INT NOT NULL,
    reading_date DATE NOT NULL DEFAULT CURRENT_DATE,
    previous_reading NUMERIC(10, 2) NOT NULL CHECK (previous_reading >= 0),
    current_reading NUMERIC(10, 2) NOT NULL CHECK (current_reading >= 0),
    volume NUMERIC(10, 2) GENERATED ALWAYS AS (current_reading - previous_reading) STORED, -- Авто-расчет объема
    CONSTRAINT fk_reading_account FOREIGN KEY (account_id) REFERENCES accounts(id) ON DELETE CASCADE,
    CONSTRAINT fk_reading_service FOREIGN KEY (service_id) REFERENCES services(id) ON DELETE CASCADE,
    CONSTRAINT chk_readings CHECK (current_reading >= previous_reading)
);

-- Таблица: Начисления
CREATE TABLE accruals (
    id SERIAL PRIMARY KEY,
    account_id INT NOT NULL,
    service_id INT NOT NULL,
    accrual_date DATE NOT NULL,
    base_amount NUMERIC(10, 2) NOT NULL CHECK (base_amount >= 0),
    discount_amount NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (discount_amount >= 0),
    penalty_amount NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (penalty_amount >= 0),
    final_amount NUMERIC(10, 2) NOT NULL CHECK (final_amount >= 0),
    is_paid BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT fk_accrual_account FOREIGN KEY (account_id) REFERENCES accounts(id) ON DELETE CASCADE,
    CONSTRAINT fk_accrual_service FOREIGN KEY (service_id) REFERENCES services(id) ON DELETE CASCADE
);

-- Таблица: Платежи
CREATE TABLE payments (
    id SERIAL PRIMARY KEY,
    accrual_id INT NOT NULL,
    payment_date DATE NOT NULL DEFAULT CURRENT_DATE,
    amount_paid NUMERIC(10, 2) NOT NULL CHECK (amount_paid > 0),
    CONSTRAINT fk_payment_accrual FOREIGN KEY (accrual_id) REFERENCES accruals(id) ON DELETE CASCADE
);
