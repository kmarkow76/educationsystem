-- 1. ТАБЛИЦА: Владельцы животных
CREATE TABLE owners (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    is_regular BOOLEAN DEFAULT FALSE NOT NULL -- Флаг "Постоянный клиент" для доп. скидки 5%
);

-- 2. ТАБЛИЦА: Животные (Электронная ветеринарная карта)
CREATE TABLE animals (
    id SERIAL PRIMARY KEY,
    pet_name VARCHAR(100) NOT NULL,
    species VARCHAR(50) NOT NULL, -- Вид (Кот, Собака, Попугай)
    breed VARCHAR(100),           -- Порода
    age_years INT NOT NULL CHECK (age_years >= 0),
    owner_id INT NOT NULL,
    CONSTRAINT fk_animal_owner FOREIGN KEY (owner_id) REFERENCES owners(id) ON DELETE CASCADE
);

-- 3. ТАБЛИЦА: Ветеринарные врачи
CREATE TABLE vets (
    id SERIAL PRIMARY KEY,
    doctor_name VARCHAR(150) NOT NULL,
    specialization VARCHAR(100) NOT NULL
);

-- 4. ТАБЛИЦА: Приемы животных
CREATE TABLE appointments (
    id SERIAL PRIMARY KEY,
    animal_id INT NOT NULL,
    vet_id INT NOT NULL,
    appointment_date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    diagnosis TEXT NOT NULL DEFAULT 'На обследовании',
    treatment TEXT, -- Назначенное лечение / процедуры
    services_cost DECIMAL(10, 2) NOT NULL CHECK (services_cost >= 0), -- Стоимость услуг
    meds_cost DECIMAL(10, 2) NOT NULL DEFAULT 0.00 CHECK (meds_cost >= 0), -- Стоимость медикаментов
    status VARCHAR(20) NOT NULL DEFAULT 'Запланирован' CHECK (status IN ('Запланирован', 'Проведен', 'Отменен')),
    CONSTRAINT fk_appointment_animal FOREIGN KEY (animal_id) REFERENCES animals(id) ON DELETE CASCADE,
    CONSTRAINT fk_appointment_vet FOREIGN KEY (vet_id) REFERENCES vets(id) ON DELETE CASCADE
);
