-- Посетители (клиенты)
CREATE TABLE Customers (
    customer_id       SERIAL PRIMARY KEY,
    full_name         VARCHAR(100) NOT NULL,
    phone             VARCHAR(20),
    email             VARCHAR(100),
    is_permanent      BOOLEAN DEFAULT FALSE,
    registration_date DATE DEFAULT CURRENT_DATE
);

-- Сотрудники
CREATE TABLE Employees (
    employee_id SERIAL PRIMARY KEY,
    full_name   VARCHAR(100) NOT NULL,
    position    VARCHAR(50),
    phone       VARCHAR(20)
);

-- Столики
CREATE TABLE Tables (
    table_id     SERIAL PRIMARY KEY,
    table_number INTEGER NOT NULL UNIQUE,
    seats        INTEGER NOT NULL,
    status       VARCHAR(20) DEFAULT 'free'   -- free, occupied, reserved
);

-- Блюда (меню)
CREATE TABLE Dishes (
    dish_id  SERIAL PRIMARY KEY,
    name     VARCHAR(100) NOT NULL,
    price    DECIMAL(10,2) NOT NULL,
    category VARCHAR(50)
);

-- Заказы (РАСШИРЕННАЯ под Задание №2 и №3)
CREATE TABLE Orders (
    order_id         SERIAL PRIMARY KEY,
    customer_id      INTEGER REFERENCES Customers(customer_id) ON DELETE SET NULL,
    table_id         INTEGER REFERENCES Tables(table_id) ON DELETE SET NULL,
    employee_id      INTEGER REFERENCES Employees(employee_id) ON DELETE SET NULL,
    order_date       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status           VARCHAR(20) DEFAULT 'created',   -- created, cooking, done, paid
    total_amount     DECIMAL(10,2) DEFAULT 0,         -- Сумма без скидки
    discount_percent INTEGER DEFAULT 0,               -- Процент скидки (из Задания №2)
    final_amount     DECIMAL(10,2) DEFAULT 0          -- Итоговая стоимость со скидкой
);

-- Детали заказа (позиции)
CREATE TABLE OrderItems (
    order_item_id  SERIAL PRIMARY KEY,
    order_id       INTEGER NOT NULL REFERENCES Orders(order_id) ON DELETE CASCADE,
    dish_id        INTEGER NOT NULL REFERENCES Dishes(dish_id) ON DELETE RESTRICT,
    quantity       INTEGER NOT NULL CHECK (quantity > 0),
    price_at_time  DECIMAL(10,2) NOT NULL   
);

-- Бронирования столиков
CREATE TABLE Reservations (
    reservation_id   SERIAL PRIMARY KEY,
    customer_id      INTEGER NOT NULL REFERENCES Customers(customer_id) ON DELETE CASCADE,
    table_id         INTEGER NOT NULL REFERENCES Tables(table_id) ON DELETE CASCADE,
    reservation_time TIMESTAMP NOT NULL,
    status           VARCHAR(20) DEFAULT 'active'   -- active, cancelled, completed
);
