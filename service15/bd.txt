-- ============================================================================
-- 1. СОЗДАНИЕ ТАБЛИЦ (3НФ и ссылочная целостность)
-- ============================================================================

-- Таблица: Клиенты
CREATE TABLE clients (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL
);

-- Таблица: Мастера
CREATE TABLE masters (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    specialty VARCHAR(100) NOT NULL, -- Сантехник, Электрик и т.д.
    rating NUMERIC(3, 2) NOT NULL CHECK (rating >= 0.00 AND rating <= 5.00)
);

-- Таблица: Виды работ / Услуги
CREATE TABLE services (
    id SERIAL PRIMARY KEY,
    service_name VARCHAR(150) NOT NULL,
    base_price NUMERIC(10, 2) NOT NULL CHECK (base_price >= 0)
);

-- Таблица: Заявки
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    client_id INT NOT NULL,
    master_id INT NOT NULL,
    service_id INT NOT NULL,
    execution_address VARCHAR(250) NOT NULL,
    execution_date DATE NOT NULL,
    order_status VARCHAR(50) NOT NULL DEFAULT 'Новая', -- Новая, В работе, Выполнена, Отменена
    CONSTRAINT fk_order_client FOREIGN KEY (client_id) REFERENCES clients(id) ON DELETE CASCADE,
    CONSTRAINT fk_order_master FOREIGN KEY (master_id) REFERENCES masters(id) ON DELETE CASCADE,
    CONSTRAINT fk_order_service FOREIGN KEY (service_id) REFERENCES services(id) ON DELETE CASCADE
);
