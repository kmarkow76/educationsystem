-- Маршруты
CREATE TABLE routes (
    id SERIAL NOT NULL PRIMARY KEY,
    route_number VARCHAR(10) NOT NULL,
    start_point VARCHAR(100) NOT NULL,
    end_point VARCHAR(100) NOT NULL,
    base_price DECIMAL(10,2) NOT NULL
);

-- Транспортные средства
CREATE TABLE vehicles (
    id SERIAL NOT NULL PRIMARY KEY,
    license_plate VARCHAR(15) NOT NULL,
    model VARCHAR(100) NOT NULL,
    capacity INT NOT NULL,
    type VARCHAR(50) NOT NULL  -- 'автобус', 'троллейбус', 'маршрутка'
);

-- Водители
CREATE TABLE drivers (
    id SERIAL NOT NULL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(16) NOT NULL,
    license_number VARCHAR(20) NOT NULL
);

-- Расписание рейсов
CREATE TABLE schedule (
    id SERIAL NOT NULL PRIMARY KEY,
    route_id INT NOT NULL REFERENCES routes(id),
    vehicle_id INT NOT NULL REFERENCES vehicles(id),
    driver_id INT NOT NULL REFERENCES drivers(id),
    departure_time TIME NOT NULL,
    arrival_time TIME NOT NULL,
    trip_date DATE NOT NULL,
    status VARCHAR(20) NOT NULL  -- 'запланирован', 'выполняется', 'завершен', 'отменен'
);

-- Пассажиры
CREATE TABLE passengers (
    id SERIAL NOT NULL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(16) NOT NULL,
    is_privileged BOOLEAN NOT NULL DEFAULT FALSE,  -- льготный: скидка 50%
    is_regular BOOLEAN NOT NULL DEFAULT FALSE       -- постоянный: скидка 10%
);

-- Билеты
CREATE TABLE tickets (
    id SERIAL NOT NULL PRIMARY KEY,
    schedule_id INT NOT NULL REFERENCES schedule(id),
    passenger_id INT NOT NULL REFERENCES passengers(id),
    ticket_type VARCHAR(20) NOT NULL,  -- 'разовый', 'месячный'
    purchase_date DATE NOT NULL
);
