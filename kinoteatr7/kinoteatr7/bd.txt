-- ========================================================
-- 1. СОЗДАНИЕ ТАБЛИЦ (Учитывая 3НФ и ссылочную целостность)
-- ========================================================

-- Таблица фильмов
CREATE TABLE movies (
    movie_id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    duration_minutes INT NOT NULL,
    genre VARCHAR(100) NOT NULL
);

-- Таблица кинозалов
CREATE TABLE halls (
    hall_id SERIAL PRIMARY KEY,
    hall_name VARCHAR(50) NOT NULL,
    total_seats INT NOT NULL
);

-- Таблица сотрудников
CREATE TABLE employees (
    employee_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    position VARCHAR(100) NOT NULL
);

-- Таблица клиентов
CREATE TABLE clients (
    client_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    is_regular INT DEFAULT 0 -- 1 - Постоянный (доп. скидка 5%), 0 - Обычный
);

-- Таблица сеансов (Связывает фильмы и залы)
CREATE TABLE sessions (
    session_id SERIAL PRIMARY KEY,
    movie_id INT NOT NULL,
    hall_id INT NOT NULL,
    session_date TIMESTAMP NOT NULL,
    base_price DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (movie_id) REFERENCES movies(movie_id) ON DELETE CASCADE,
    FOREIGN KEY (hall_id) REFERENCES halls(hall_id) ON DELETE CASCADE
);

-- Таблица билетов (Учет продаж и бронирования)
CREATE TABLE tickets (
    ticket_id SERIAL PRIMARY KEY,
    session_id INT NOT NULL,
    client_id INT NOT NULL,
    employee_id INT NOT NULL,
    row_number INT NOT NULL,
    seat_number INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1,
    status VARCHAR(50) NOT NULL, -- 'Свободно', 'Забронировано', 'Продано'
    FOREIGN KEY (session_id) REFERENCES sessions(session_id) ON DELETE CASCADE,
    FOREIGN KEY (client_id) REFERENCES clients(client_id) ON DELETE CASCADE,
    FOREIGN KEY (employee_id) REFERENCES employees(employee_id) ON DELETE CASCADE
);
