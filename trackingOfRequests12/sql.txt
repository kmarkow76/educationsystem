-- Клиенты
CREATE TABLE clients (
    id SERIAL NOT NULL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(16) NOT NULL,
    is_regular BOOLEAN NOT NULL DEFAULT FALSE  -- постоянный клиент: скидка 5%
);

-- Сотрудники
CREATE TABLE employees (
    id SERIAL NOT NULL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(16) NOT NULL,
    position VARCHAR(100) NOT NULL  -- должность
);

-- Техника
CREATE TABLE devices (
    id SERIAL NOT NULL PRIMARY KEY,
    client_id INT NOT NULL REFERENCES clients(id),
    device_type VARCHAR(100) NOT NULL,   -- 'компьютер', 'телефон', 'бытовая техника'
    brand VARCHAR(100) NOT NULL,
    model VARCHAR(100) NOT NULL,
    serial_number VARCHAR(50) NOT NULL
);

-- Запасные части
CREATE TABLE spare_parts (
    id SERIAL NOT NULL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    quantity_in_stock INT NOT NULL
);

-- Заявки на ремонт
CREATE TABLE repair_requests (
    id SERIAL NOT NULL PRIMARY KEY,
    client_id INT NOT NULL REFERENCES clients(id),
    device_id INT NOT NULL REFERENCES devices(id),
    employee_id INT NOT NULL REFERENCES employees(id),
    fault_description TEXT NOT NULL,   -- описание неисправности
    work_list TEXT NOT NULL,           -- перечень работ
    base_work_price DECIMAL(10,2) NOT NULL,  -- базовая стоимость работ
    is_urgent BOOLEAN NOT NULL DEFAULT FALSE, -- срочность: +20%
    status VARCHAR(20) NOT NULL,       -- 'принята', 'в ремонте', 'выполнена', 'выдана'
    created_date DATE NOT NULL
);

-- Запасные части использованные в заявке
CREATE TABLE request_parts (
    id SERIAL NOT NULL PRIMARY KEY,
    request_id INT NOT NULL REFERENCES repair_requests(id),
    part_id INT NOT NULL REFERENCES spare_parts(id),
    quantity INT NOT NULL
);
