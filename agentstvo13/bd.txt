-- 1. СОЗДАНИЕ ТАБЛИЦ (3НФ и обеспечение ссылочной целостности)
-- ============================================================================

-- Таблица: Клиенты
CREATE TABLE clients (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    is_repeat BOOLEAN NOT NULL DEFAULT FALSE -- Флаг: повторный клиент (скидка 5%)
);

-- Таблица: Площадки
CREATE TABLE venues (
    id SERIAL PRIMARY KEY,
    venue_name VARCHAR(150) NOT NULL,
    address VARCHAR(250) NOT NULL,
    rental_price NUMERIC(10, 2) NOT NULL CHECK (rental_price >= 0)
);

-- Таблица: Подрядчики / Услуги
CREATE TABLE contractors (
    id SERIAL PRIMARY KEY,
    contractor_name VARCHAR(150) NOT NULL,
    service_type VARCHAR(100) NOT NULL, -- Фотограф, Ведущий, Декор и т.д.
    service_cost NUMERIC(10, 2) NOT NULL CHECK (service_cost >= 0)
);

-- Таблица: Мероприятия
CREATE TABLE events (
    id SERIAL PRIMARY KEY,
    event_name VARCHAR(150) NOT NULL,
    contract_date DATE NOT NULL DEFAULT CURRENT_DATE, -- Дата заключения договора
    event_date DATE NOT NULL,                         -- Дата проведения
    payment_status VARCHAR(50) NOT NULL DEFAULT 'Не оплачено', -- Не оплачено, Предоплата, Оплачено
    client_id INT NOT NULL,
    venue_id INT NOT NULL,
    CONSTRAINT fk_event_client FOREIGN KEY (client_id) REFERENCES clients(id) ON DELETE CASCADE,
    CONSTRAINT fk_event_venue FOREIGN KEY (venue_id) REFERENCES venues(id) ON DELETE CASCADE,
    CONSTRAINT chk_event_dates CHECK (event_date >= contract_date)
);

-- Таблица: Детали сметы (Разделение связи многие-ко-многим)
CREATE TABLE event_details (
    id SERIAL PRIMARY KEY,
    event_id INT NOT NULL,
    contractor_id INT NOT NULL,
    CONSTRAINT fk_details_event FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE,
    CONSTRAINT fk_details_contractor FOREIGN KEY (contractor_id) REFERENCES contractors(id) ON DELETE CASCADE
);
