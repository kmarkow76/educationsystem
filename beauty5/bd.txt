-- 1. Таблица Клиентов (Соблюдена 3НФ, есть флаг постоянного клиента для скидки 5%)
CREATE TABLE clients (
    client_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    is_regular INT NOT NULL DEFAULT 0 -- 1 - постоянный (скидка 5%), 0 - обычный
);

-- 2. Таблица Мастеров
CREATE TABLE masters (
    master_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    specialization VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL
);

-- 3. Таблица Услуг (Прайс-лист)
CREATE TABLE services (
    service_id SERIAL PRIMARY KEY,
    service_name VARCHAR(150) NOT NULL,
    price DECIMAL(10, 2) NOT NULL
);

-- 4. Таблица Записей (Журнал визитов)
CREATE TABLE appointments (
    appointment_id SERIAL PRIMARY KEY,
    client_id INT NOT NULL,
    master_id INT NOT NULL,
    appointment_date TIMESTAMP NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Запланирована', -- Запланирована, Выполнена, Отменена
    FOREIGN KEY (client_id) REFERENCES clients(client_id) ON DELETE CASCADE,
    FOREIGN KEY (master_id) REFERENCES masters(master_id) ON DELETE CASCADE
);

-- 5. Промежуточная таблица: Детали записи (Содержит перечень оказанных услуг в визите)
CREATE TABLE appointment_details (
    detail_id SERIAL PRIMARY KEY,
    appointment_id INT NOT NULL,
    service_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1,
    FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id) ON DELETE CASCADE,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);
