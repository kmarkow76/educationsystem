-- 1. Справочник тарифов (Почасовые и фиксированные пакеты)
CREATE TABLE tariffs (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL,                -- 'Standard', 'VIP', 'Пакет Ночь'
    price_per_hour DECIMAL(10, 2) NOT NULL,   -- Базовая цена за час
    is_night_package BOOLEAN DEFAULT FALSE    -- Флаг: является ли тарифом "Ночь"
);

-- 2. Игровые зоны (ПК, Консоли, VIP-комнаты)
CREATE TABLE game_zones (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL,                -- 'PC Common', 'VIP Room', 'PS5 Zone'
    description TEXT
);

-- 3. Игровые места / Устройства (Компьютеры и консоли)
CREATE TABLE gaming_places (
    id SERIAL PRIMARY KEY,
    zone_id INT NOT NULL REFERENCES game_zones(id) ON DELETE RESTRICT,
    place_number INT NOT NULL UNIQUE,          -- Уникальный номер места/ПК в клубе
    hardware_spec TEXT,                       -- Характеристики (RTX 4060, 16GB)
    is_occupied BOOLEAN DEFAULT FALSE         -- Текущий статус занятости места
);

-- 4. Участники клуба (Клиенты)
CREATE TABLE club_members (
    id SERIAL PRIMARY KEY,
    nickname VARCHAR(50) NOT NULL UNIQUE,     -- Игровой никнейм (уникальный)
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    has_club_card BOOLEAN DEFAULT FALSE       -- Наличие клубной карты (+10% скидка)
);

-- 5. Игровые сессии (Учет времени и аренды мест)
CREATE TABLE game_sessions (
    id SERIAL PRIMARY KEY,
    member_id INT NOT NULL REFERENCES club_members(id) ON DELETE CASCADE,
    place_id INT NOT NULL REFERENCES gaming_places(id) ON DELETE RESTRICT,
    tariff_id INT NOT NULL REFERENCES tariffs(id) ON DELETE RESTRICT,
    start_time TIMESTAMP NOT NULL,
    end_time TIMESTAMP,                       -- NULL, пока сессия активна
    base_price DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    discount_percent INT NOT NULL DEFAULT 0,  -- Рассчитанный суммарный % скидки
    total_price DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    status VARCHAR(20) NOT NULL DEFAULT 'Активна' -- 'Активна', 'Завершена', 'Пауза'
);

-- 6. Справочник товаров бара
CREATE TABLE bar_products (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    quantity_in_stock INT NOT NULL DEFAULT 0
);

-- 7. Продажи из бара в рамках игровых сессий
CREATE TABLE bar_sales (
    id SERIAL PRIMARY KEY,
    session_id INT NOT NULL REFERENCES game_sessions(id) ON DELETE CASCADE,
    product_id INT NOT NULL REFERENCES bar_products(id) ON DELETE RESTRICT,
    quantity INT NOT NULL DEFAULT 1,
    sale_price DECIMAL(10, 2) NOT NULL        -- Фиксация цены на момент продажи
);
